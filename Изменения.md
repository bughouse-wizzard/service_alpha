    async def get_search_results(
        self,
        search_id: uuid.UUID,
        accepted_for_nmc: Optional[bool] = None,
        is_2025_plus: Optional[bool] = None,
        manufacturer_match: Optional[bool] = None,
        page: int = 1,
        page_size: int = 50
    ) -> SearchResultListResponse:
        """Получает результаты поиска с фильтрацией."""
        # Применяем фильтры для репозитория
        filters = {}
        if accepted_for_nmc is not None:
            filters["accepted_for_nmc"] = accepted_for_nmc
        if is_2025_plus is not None:
            filters["is_2025_plus"] = is_2025_plus
        if manufacturer_match is not None:
            filters["manufacturer_match"] = manufacturer_match

        total, results = await self.result_repo.list_results(
            self.db,
            search_id=search_id,
            page=page,
            page_size=page_size,
            **filters
        )
        # Формируем список результатов с нужными полями (contract_id, published_at, price, etc.)
        items = []
        for res in results:
            # Имя производителя (найденного) и заказчика из данных
            customer_name = res.raw_data_json.get("customer") if res.raw_data_json else None
            items.append({
                "contract_id": res.reestr_number,                        # Реестровый номер контракта
                "published_at": res.sign_date.isoformat() if res.sign_date else None,  # Дата контракта
                "price": float(res.unit_price) if res.unit_price is not None else None,  # Итоговая цена контракта (равна unit_price при отсутствии отдельного поля)
                "unit_price": float(res.unit_price) if res.unit_price is not None else None,
                "currency": res.currency,
                "customer": customer_name or "",
                "manufacturer": res.manufacturer_found or "",
                "match_type": res.match_type,
                "is_identical": True if res.match_type == "IDENTICAL" else False,
                "ai_score": res.ai_score or 0.0,
                "manufacturer_match": bool(res.manufacturer_match),
                "is_2025_plus": bool(res.is_2025_plus),
                "accepted_for_nmc": bool(res.accepted_for_nmc)
            })
        return SearchResultListResponse(
            search_id=str(search_id),
            page=page,
            page_size=page_size,
            total=total,
            items=items
        )
Method SearchService.update_selection – Accept reestr numbers and update selection
    async def update_selection(self, search_id: str, selected_numbers: list[str]) -> SearchDetailsResponse | None:
        """Обновляет выбор контрактов для НМЦК на основе реестровых номеров."""
        sid = uuid.UUID(search_id)
        search = await self.search_repo.get(self.db, sid)
        if not search:
            return None

        # Множество выбранных реестровых номеров (до 3)
        selected_set = set(selected_numbers)
        # Находим соответствующие ID контрактов в этом поиске по реестровым номерам
        if selected_set:
            query = sa.select(ContractResult.id, ContractResult.reestr_number).where(
                ContractResult.search_id == sid,
                ContractResult.reestr_number.in_(selected_set)
            )
            result = await self.db.execute(query)
            found_pairs = result.all()
            found_ids = {row[0] for row in found_pairs}
            found_numbers = {row[1] for row in found_pairs}
            # Проверяем, есть ли номера, не принадлежащие данному поиску
            invalid = selected_set - found_numbers
            if invalid:
                raise ValueError(f"Некорректные реестровые номера: {list(invalid)}")

            # Собираем цены выбранных контрактов для расчета НМЦК
            prices_query = sa.select(ContractResult.unit_price).where(
                ContractResult.search_id == sid,
                ContractResult.reestr_number.in_(selected_set)
            )
            prices_result = await self.db.execute(prices_query)
            prices = [row[0] for row in prices_result.all() if row[0] is not None]
            nmc_value = calc_nmc_avg(prices) if prices else None
        else:
            found_ids = set()
            nmc_value = None

        # Сбрасываем флаг accepted_for_nmc для всех результатов поиска
        await self.db.execute(
            sa.update(ContractResult)
            .where(ContractResult.search_id == sid)
            .values(accepted_for_nmc=False)
        )
        # Устанавливаем флаг для выбранных результатов
        if found_ids:
            await self.db.execute(
                sa.update(ContractResult)
                .where(ContractResult.search_id == sid, ContractResult.id.in_(found_ids))
                .values(accepted_for_nmc=True)
            )
        # Обновляем запись поиска (список выбранных ID и новое NMC значение)
        search.selected_contract_ids = list(found_ids)
        search.nmc_value = nmc_value
        await self.search_repo.update(self.db, search)
        await self.db.commit()

        # Эмитим SSE-событие об обновлении выбора (кол-во и новое значение НМЦК)
        await self.broker.emit(
            str(search_id),
            StreamEvent("selection", {
                "search_id": str(search_id),
                "selected_count": len(found_ids),
                "nmc_value": str(nmc_value) if nmc_value is not None else None
            })
        )
        log().info(
            "selection_updated",
            search_id=str(search_id),
            selected_count=len(found_ids),
            nmc_value=str(nmc_value) if nmc_value else None
        )
        # Возвращаем обновленные детали поиска
        return self._to_details(search)

async def get_comparison_data(
    search_id: UUID,
    contract_id: str,
    db: AsyncSession = Depends(get_db)
) -> ComparisonResponse:
    """Получить данные сравнения характеристик для указанного контракта."""
    # Ищем результат по ID или реестровому номеру в рамках данного поиска
    query = (
        sa.select(ContractResult)
        .where(ContractResult.search_id == search_id)
        .where(
            sa.or_(
                ContractResult.id == contract_id,                # если UUID
                ContractResult.reestr_number == contract_id      # если строка номера
            )
        )
    )
    result = (await db.execute(query)).scalar_one_or_none()
    if not result:
        raise ApiException(
            status_code=status.HTTP_404_NOT_FOUND,
            error_code="NOT_FOUND",
            message="Contract result not found",
            details={"contract_id": contract_id}
        )
    # Получаем строки сравнения характеристик (если есть)
    rows = []
    for row in result.comparison_rows:  # relationship "comparison_rows" is loaded on access
        rows.append({
            "name": row.name,
            "target": row.target_value,
            "actual": row.actual_value,
            "status": row.match_status,
            "reason": row.reason or ""
        })
    return ComparisonResponse(
        result_id=str(result.id),
        contract_id=result.reestr_number,
        rows=rows
    )

async function loadSearchHistory() {
    try {
        const response = await fetch('/api/search?page=1&page_size=50');
        if (!response.ok) throw new Error('Не удалось загрузить историю поисков');
        const data = await response.json();
        // Ожидаем объект с полем items (список SearchBrief)
        SEARCH_HISTORY = data.items.map(item => ({
            id: item.id,
            date: new Date(item.created_at).toISOString().split('T')[0],
            name: item.object_name || 'Не указано',
            ktru: item.ktru,
            status: mapStatus(item.status),
            found: item.found_total || 0,
            avg: item.nmc_value ? parseFloat(item.nmc_value) : null,
            type: "generic"
        }));
        // Сортируем по дате создания (новые записи сверху)
        SEARCH_HISTORY.sort((a, b) => new Date(b.date) - new Date(a.date));
    } catch (error) {
        console.error('Ошибка загрузки истории:', error);
        showToast('Не удалось загрузить историю поисков', 'error');
        SEARCH_HISTORY = [];
    }
}

Function setupSSE(searchId) – Handle updated SSE events and statuses
function setupSSE(searchId) {
    if (eventSource) {
        eventSource.close();
    }
    eventSource = new EventSource(`${API_BASE}/search/${searchId}/events`);
    eventSource.onopen = () => {
        console.log(`SSE подключен для поиска ${searchId}`);
    };
    eventSource.addEventListener('progress', (event) => {
        const data = JSON.parse(event.data);
        const item = SEARCH_HISTORY.find(i => i.id === searchId);
        if (item) {
            item.found = data.found_total || 0;
            item.processed_count = data.processed_count || 0;
            if (activeSidebarSearchId === searchId) {
                const progress = item.limit ? Math.min(100, (item.processed_count / item.limit) * 100) : 0;
                document.getElementById('proc-bar').style.width = `${progress}%`;
                document.getElementById('proc-count').innerText = `${item.processed_count}/${item.limit}`;
            }
            renderHistory();
        }
    });
    eventSource.addEventListener('result_added', async (event) => {
        const data = JSON.parse(event.data);
        console.log('Новый результат:', data);
        // Если пользователь на вкладке "Результаты" данного поиска, обновляем таблицу
        const currentSearch = SEARCH_HISTORY.find(i => i.id === searchId);
        if (currentSearch && document.getElementById('tab-results').classList.contains('active')) {
            await loadResultsFromAPI(searchId);
        }
    });
    eventSource.addEventListener('done', async (event) => {
        const data = JSON.parse(event.data);
        const item = SEARCH_HISTORY.find(i => i.id === searchId);
        if (item) {
            // Устанавливаем корректный финальный статус по коду
            item.status = data.status ? mapStatus(data.status) : 'Завершен';
            item.found = data.found_total || 0;
            item.avg = data.nmc_value ? parseFloat(data.nmc_value) : 0;
            showToast(`Поиск завершен. Найдено ${item.found} контрактов`, data.status === 'failed' ? 'error' : 'success');
            if (activeSidebarSearchId === searchId) {
                resetSidebarUI();
            }
            renderHistory();
            // Закрываем SSE после завершения
            eventSource.close();
            eventSource = null;
        }
    });
    eventSource.addEventListener('error', (event) => {
        console.error('SSE ошибка:', event);
        showToast('Ошибка подключения к серверу событий', 'error');
        if (eventSource) {
            eventSource.close();
            eventSource = null;
        }
    });
}
