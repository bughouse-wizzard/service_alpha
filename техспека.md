7) ORM под PostgreSQL: SQLAlchemy 2.0 async (конкретика)
7.1. Сессии и транзакции — правило
·	В API слой передаём AsyncSession через DI
·	Транзакция на use-case (или на репозиторий — но единообразно)
·	Никаких “глобальных сессий”
Пример engine + sessionmaker
from sqlalchemy.ext.asyncio import AsyncSession, create_async_engine
from sqlalchemy.orm import sessionmaker

DATABASE_URL = "postgresql+asyncpg://app:pass@db:5432/appdb"

engine = create_async_engine(
    DATABASE_URL,
    pool_size=10,
    max_overflow=20,
    pool_pre_ping=True,
)

AsyncSessionLocal = sessionmaker(
    bind=engine,
    class_=AsyncSession,
    expire_on_commit=False,
)

Зависимость FastAPI
from typing import AsyncGenerator

async def get_db() -> AsyncGenerator[AsyncSession, None]:
    async with AsyncSessionLocal() as session:
        yield session

7.2. Модели SQLAlchemy (2.0 style) + JSONB
from sqlalchemy.orm import DeclarativeBase, Mapped, mapped_column
from sqlalchemy.dialects.postgresql import JSONB
from sqlalchemy import String, Integer, Boolean, ForeignKey, UniqueConstraint
from uuid import uuid4
import sqlalchemy as sa

class Base(DeclarativeBase):
    pass

class SearchRequest(Base):
    __tablename__ = "search_request"
    id: Mapped[str] = mapped_column(String, primary_key=True, default=lambda: str(uuid4()))
    status: Mapped[str] = mapped_column(String, index=True)
    ktru: Mapped[str] = mapped_column(String, index=True)
    found_total: Mapped[int] = mapped_column(Integer, default=0)
    processed_count: Mapped[int] = mapped_column(Integer, default=0)
    created_at: Mapped[sa.DateTime] = mapped_column(sa.DateTime(timezone=True), server_default=sa.func.now())

class ContractResult(Base):
    __tablename__ = "contract_result"
    __table_args__ = (
        UniqueConstraint("search_id", "reestr_number", name="uq_search_contract"),
    )

    id: Mapped[str] = mapped_column(String, primary_key=True, default=lambda: str(uuid4()))
    search_id: Mapped[str] = mapped_column(ForeignKey("search_request.id"), index=True)
    reestr_number: Mapped[str] = mapped_column(String, index=True)
    ai_score: Mapped[int] = mapped_column(Integer, default=0, index=True)
    match_type: Mapped[str] = mapped_column(String, index=True)
    accepted_for_nmc: Mapped[bool] = mapped_column(Boolean, default=False)
    raw_data_json: Mapped[dict] = mapped_column(JSONB, default=dict)

7.3. Репозитории (Repository pattern)
from sqlalchemy import select
from sqlalchemy.ext.asyncio import AsyncSession

class SearchRepo:
    async def get(self, db: AsyncSession, search_id: str) -> SearchRequest:
        res = await db.execute(select(SearchRequest).where(SearchRequest.id == search_id))
        obj = res.scalar_one()
        return obj

    async def upsert_contract_result(self, db: AsyncSession, result: ContractResult) -> None:
        # Вариант: INSERT ... ON CONFLICT DO UPDATE (Postgres) через SQLAlchemy dialect
        db.add(result)

Важно: для идемпотентности и дедупа под Postgres используем ON CONFLICT стратегию (upsert) для (search_id, reestr_number).
